cmake_minimum_required(VERSION 3.22)
project(shogi_oss_ai CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(variant full light)
 set(root "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/sunfish-${variant}")
 file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${root}/src/core/*.cpp" "${root}/src/search/*.cpp" "${root}/src/common/*.cpp" "${root}/src/logger/*.cpp")
 add_library(sunfish_${variant} SHARED sunfish_bridge.cpp ${sources})
 target_include_directories(sunfish_${variant} PRIVATE "${root}/src")
 target_compile_definitions(sunfish_${variant} PRIVATE UNIX=1)
 target_compile_options(sunfish_${variant} PRIVATE -O3 -DNDEBUG -fno-rtti -fvisibility=hidden)
 target_link_options(sunfish_${variant} PRIVATE "-Wl,-z,max-page-size=16384")
 if(variant STREQUAL "light")
  target_compile_definitions(sunfish_${variant} PRIVATE SUNFISH_LIGHT=1)
 endif()
endforeach()
